  add_subdirectory(vendor/llama.cpp EXCLUDE_FROM_ALL)

    configure_file("${CMAKE_SOURCE_DIR}/vendor/${dep}.h" "${CMAKE_BINARY_DIR}/deps/${dep}.h" COPYONLY)
  endif()
endforeach()
function(pocket_header name url hash)
  if(NOT EXISTS "${CMAKE_BINARY_DIR}/deps/${name}")
    file(DOWNLOAD "${url}" "${CMAKE_BINARY_DIR}/deps/${name}" STATUS status TLS_VERIFY ON EXPECTED_HASH SHA256=${hash})
    list(GET status 0 code)
    if(NOT code EQUAL 0)
      file(REMOVE "${CMAKE_BINARY_DIR}/deps/${name}")
      message(FATAL_ERROR "Dependency download failed: ${status}")
    endif()
  endif()
endfunction()
